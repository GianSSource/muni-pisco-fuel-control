CREATE TABLE `fuel_delivery_lot_allocations` (
  `id` int(10) UNSIGNED NOT NULL,
  `delivery_id` int(10) UNSIGNED NOT NULL,
  `fuel_income_id` int(10) UNSIGNED NOT NULL,
  `gallons_taken` decimal(10,2) NOT NULL,
  `created_at` timestamp NOT NULL DEFAULT current_timestamp()
) ;

DELIMITER $$
CREATE TRIGGER `trg_alloc_after_delete` AFTER DELETE ON `fuel_delivery_lot_allocations` FOR EACH ROW BEGIN
    UPDATE fuel_incomes
    SET gallons_remaining = gallons_remaining + OLD.gallons_taken
    WHERE id = OLD.fuel_income_id;
END
$$
DELIMITER ;
DELIMITER $$
CREATE TRIGGER `trg_alloc_after_insert` AFTER INSERT ON `fuel_delivery_lot_allocations` FOR EACH ROW BEGIN
    UPDATE fuel_incomes
    SET gallons_remaining = gallons_remaining - NEW.gallons_taken
    WHERE id = NEW.fuel_income_id;
END
$$
DELIMITER ;
DELIMITER $$
CREATE TRIGGER `trg_alloc_before_insert` BEFORE INSERT ON `fuel_delivery_lot_allocations` FOR EACH ROW BEGIN
    DECLARE available DECIMAL(10,2);

    SELECT gallons_remaining INTO available
        FROM fuel_incomes WHERE id = NEW.fuel_income_id FOR UPDATE;

    IF available IS NULL THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'El lote de combustible indicado no existe';
    ELSEIF available < NEW.gallons_taken THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'No hay suficiente combustible disponible en ese lote';
    END IF;
END
$$
DELIMITER ;
